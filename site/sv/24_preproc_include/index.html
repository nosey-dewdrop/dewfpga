design.sv

`include "defs.svh"
`define USE_FEATURE
module top(input logic [15:0] sw, output logic [15:0] led);
  assign led[`WIDTH-1:0] = `MAX(sw[3:0], sw[7:4]);
`ifdef USE_FEATURE
  assign led[4 +: `WIDTH] = 4'hC;
`else
  assign led[7:4] = 4'h3;
`endif
`ifndef NOT_DEFINED
  assign led[15:8] = sw[15:8];
`endif
endmodule

defs.svh

`ifndef DEFS_SVH
`define DEFS_SVH
`define WIDTH 4
`define MAX(a,b) ((a) > (b) ? (a) : (b))
`endif

  reg clk = 1'b0;
  reg btnC = 1'b0, btnU = 1'b0, btnD = 1'b0, btnL = 1'b0, btnR = 1'b0;

  initial begin #200000 $display("FAIL: timeout"); $finish; end

  initial begin
    sw = 16'hAB29; #1
    if (!(led === 16'hABC9)) begin $display("FAIL: include/define/ifdef/+: (led=%h seg=%b an=%b)", led, seg, an); $finish; end
    $display("PASS"); $finish;
  end
endmodule
