What is the probe's code?

From test/sv/95_empty_submodule.

design.sv

// a lab mid-way: part 2's module is declared and instantiated, its body not written yet
module blinker(input logic clk, output logic led);
  // part 2 of the lab: not written yet
endmodule
module top(input logic clk, input logic [15:0] sw, output logic [15:0] led);
  logic b;
  blinker u_blink(.clk(clk), .led(b));
  assign led = {sw[15:1], b};
endmodule

tb.sv

module tb;
  // part 1, the switches on led[15:1], is what is written so far; led[0] waits for part 2
  logic clk = 0; logic [15:0] sw = 0; logic [15:0] led;
  top dut(.clk(clk), .sw(sw), .led(led));
  always #5 clk = ~clk;
  integer i, bad = 0;
  initial begin
    for (i = 0; i < 300; i++) begin
      sw = (i < 16) ? 16'(1 << i) : $random;
      @(negedge clk); if (led[15:1] !== sw[15:1]) begin bad++; if (bad < 4) $display("FAIL: sw=%h led=%h", sw, led); end
    end
    if (bad == 0) $display("PASS"); else $display("FAIL: %0d of 300", bad);
    $finish;
  end
endmodule

previous: a variable raised to a constant power: assign led = sw[7:0] ** 2;
next: a port declared [N-1:0] with N = 1: a vector of one bit, numbered 0
