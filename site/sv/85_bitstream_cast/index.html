What is the probe's code?

From test/sv/85_bitstream_cast.

design.sv

module top(input logic [15:0] sw, output logic [15:0] led);
  typedef logic [3:0] quad_t [0:1];      // an unpacked array of two nibbles
  quad_t q;
  assign q = quad_t'(sw[7:0]);           // a bitstream cast: q[0] gets sw[7:4], q[1] gets sw[3:0]
  assign led = {sw[15:8], q[1], q[0]};
endmodule

tb.sv

module tb;
  logic [15:0] sw; logic [15:0] led;
  top dut(.sw(sw), .led(led));
  integer i, bad = 0;
  initial begin
    for (i = 0; i < 256; i++) begin
      sw = {i[7:0] ^ 8'hc3, i[7:0]}; #1;
      if (led !== {sw[15:8], sw[3:0], sw[7:4]}) begin bad++; if (bad < 4) $display("FAIL: sw=%h led=%h", sw, led); end
    end
    if (bad == 0) $display("PASS"); else $display("FAIL: %0d of 256", bad);
    $finish;
  end
endmodule
