What is the probe's code?

From test/sv/61_enum_range.

design.sv

module top(input logic [15:0] sw, output logic [15:0] led);
  typedef enum logic [1:0] {S[3]} s_t;        // an enum range: S0, S1, S2 (values 0, 1, 2)
  typedef enum logic [2:0] {T[5:7]} t_t;      // T5, T6, T7 (values 0, 1, 2)
  s_t s; t_t t;
  always_comb
    case (sw[1:0])
      2'd0:    s = S2;
      2'd1:    s = S0;
      default: s = S1;
    endcase
  always_comb
    if (sw[2]) t = T7;
    else       t = T6;
  assign led = {11'd0, t, s};
endmodule

tb.sv

module tb;
  reg [15:0] sw = 16'h0000;
  wire [15:0] led;
  integer i; reg [1:0] s; reg [2:0] t;
  top dut(.sw(sw), .led(led));
  initial begin
    for (i = 0; i < 8; i = i + 1) begin
      sw = {13'h1abc, i[2:0]}; #1
      s = (sw[1:0] == 2'd0) ? 2'd2 : (sw[1:0] == 2'd1) ? 2'd0 : 2'd1;
      t = sw[2] ? 3'd2 : 3'd1;
      if (!(led === {11'd0, t, s})) begin $display("FAIL: S0..S2 and T5..T7 values (sw=%h led=%h)", sw, led); $finish; end
    end
    $display("PASS"); $finish;
  end
endmodule
