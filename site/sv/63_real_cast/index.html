What is the probe's code?

From test/sv/63_real_cast.

design.sv

module top(input logic [15:0] sw, output logic [15:0] led);
  localparam real STEP = 2.6;           // a real parameter
  localparam int INC = int'(STEP);      // cast to int: rounds to 3
  assign led = sw + INC;
endmodule

tb.sv

module tb;
  reg [15:0] sw = 16'h0000;
  wire [15:0] led;
  integer i;
  top dut(.sw(sw), .led(led));
  initial begin
    for (i = 0; i < 65536; i = i + 1) begin
      sw = i; #1
      if (!(led === 16'(sw + 16'd3))) begin $display("FAIL: int'(2.6) is 3 (sw=%h led=%h)", sw, led); $finish; end
    end
    $display("PASS"); $finish;
  end
endmodule
