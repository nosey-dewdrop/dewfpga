What is the probe's code?

From test/sv/46_continue.

design.sv

module top(input logic [15:0] sw, output logic [15:0] led);
  logic [4:0] n;
  always_comb begin
    n = 0;
    for (int i = 0; i < 16; i++) begin
      if (!sw[i]) continue;   // skip the switches that are off
      n++;
    end
  end
  assign led = {11'b0, n};
endmodule

  wire [15:0] led; wire [6:0] seg; wire dp; wire [3:0] an;

  initial begin #200000 $display("FAIL: timeout"); $finish; end
  integer i, b, c;
  top dut(.sw(sw), .led(led));
  initial begin
    for (i = 0; i < 65536; i = i + 1) begin
      sw = i; #1
      c = 0; for (b = 0; b < 16; b = b + 1) c = c + sw[b];
      if (!(led === c)) begin $display("FAIL: continue, count of set bits (sw=%h led=%h want %0d)", sw, led, c); $finish; end
    end
    $display("PASS"); $finish;
  end
endmodule
