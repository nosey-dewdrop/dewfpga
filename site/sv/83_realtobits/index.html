What is the probe's code?

From test/sv/83_realtobits.

design.sv

module top(input logic [15:0] sw, output logic [15:0] led);
  localparam logic [63:0] B = $realtobits(2.5);                 // IEEE 754: 64'h4004_0000_0000_0000
  localparam real         R = $bitstoreal(64'h3FF8_0000_0000_0000);   // 1.5
  assign led = sw ^ B[63:48] ^ 16'($rtoi(R * 4.0));             // sw ^ 16'h4004 ^ 16'd6
endmodule

tb.sv

module tb;
  logic [15:0] sw; logic [15:0] led;
  top dut(.sw(sw), .led(led));
  integer i, bad = 0;
  initial begin
    for (i = 0; i < 256; i++) begin
      sw = {i[7:0], i[7:0]}; #1;
      if (led !== (sw ^ 16'h4002)) begin bad++; if (bad < 4) $display("FAIL: sw=%h led=%h", sw, led); end
    end
    if (bad == 0) $display("PASS"); else $display("FAIL: %0d of 256", bad);
    $finish;
  end
endmodule
