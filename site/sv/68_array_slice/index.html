What is the probe's code?

From test/sv/68_array_slice.

design.sv

module top(input logic [15:0] sw, output logic [15:0] led);
  logic [3:0] a [0:3], b [0:1];
  always_comb begin
    for (int i = 0; i < 4; i++) a[i] = sw[4*i +: 4];
    b = a[2:3];                          // a slice of an unpacked array
    led = {8'd0, b[1], b[0]};
  end
endmodule

tb.sv

module tb;
  reg [15:0] sw = 16'h0000;
  wire [15:0] led;
  integer i;
  top dut(.sw(sw), .led(led));
  initial begin
    for (i = 0; i < 256; i = i + 1) begin
      sw = {i[7:0], ~i[7:0]}; #1
      if (!(led === {8'd0, sw[15:8]})) begin $display("FAIL: b = a[2:3] (sw=%h led=%h)", sw, led); $finish; end
    end
    $display("PASS"); $finish;
  end
endmodule

previous: equality of two whole unpacked arrays: a == b, a != b
next: disable of a named block to leave a loop (the lowest switch that is up)
