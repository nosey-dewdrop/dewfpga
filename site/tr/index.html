kart

      
assign led = sw çalıştıran bir basys 3. bir anahtarı çevir ya da bir lede tıkla.


module top(input  logic [15:0] sw,
           output logic [15:0] led,
           output logic [6:0]  seg,
           output logic [3:0]  an);
    assign led = sw;
    // 0 segmenti de haneyi de yakar
    assign seg = ~sw[6:0];
    assign an  = ~sw[15:12];
endmodule
